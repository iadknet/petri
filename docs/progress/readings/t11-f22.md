Probe source:

```rust
//! TEMPORARY PROBE (not for commit): T11.F22 Task 1. Reproduces the research
//! note's cheap-proof `InputRef.Swap` trials (same seeds and selector as
//! `neighborhood::operators::per_operator_rows`) on dumped live genomes under
//! the live mutation config, and records for every applied pre-change swap
//! whether the old and new entries share a kind (`input_ref_kind`).
//! `c` = cross-kind changed trials / all changed trials; `B = 0.125 + 0.259 c`.
use std::collections::BTreeMap;

use rand::rngs::SmallRng;
use rand::SeedableRng;
use v3_core::config::SimulationConfig;
use v3_core::creature::genome::analysis::mesh_reachable_nodes;
use v3_core::creature::genome::CreatureGenome;
use v3_core::mutation::input_ref::{input_ref_kind, InputRefMutator, InputRefOperator};
use v3_core::mutation::reachability::TargetSets;
use v3_core::neighborhood::operators::INPUT_REF_SEED_BASE;
use v3_core::neighborhood::{classify, Battery, Class, EvalContext};

#[derive(Default, Debug)]
struct Cell {
    silent: u32,
    changed: u32,
    dead: u32,
}

#[test]
fn probe_swap_kind() {
    let dir = std::env::var("PETRI_GENOME_DIR").expect("PETRI_GENOME_DIR");
    let config_path = std::env::var("PETRI_CONFIG").expect("PETRI_CONFIG");
    let n: usize = std::env::var("PETRI_N").ok().and_then(|v| v.parse().ok()).unwrap_or(400);
    let trials: u32 = std::env::var("PETRI_TRIALS").ok().and_then(|v| v.parse().ok()).unwrap_or(20);
    let cfg_json: serde_json::Value =
        serde_json::from_str(&std::fs::read_to_string(&config_path).unwrap()).unwrap();
    let config: SimulationConfig = serde_json::from_value(cfg_json["config"].clone()).unwrap();
    let food_type_count = config.world.food.types.len();
    let battery = Battery::generate(food_type_count);
    let ctx = EvalContext::from_config(&config);
    let mut files: Vec<_> = std::fs::read_dir(&dir)
        .unwrap()
        .filter_map(|e| e.ok())
        .map(|e| e.path())
        .filter(|p| p.extension().is_some_and(|x| x == "json"))
        .collect();
    files.sort();

    let mut applied = 0u32;
    let mut skipped = 0u32;
    let mut identical = 0u32; // redraw produced the very same reference
    let mut within = Cell::default();
    let mut cross = Cell::default();
    // (old kind, new kind) -> outcome cell
    let mut by_pair: BTreeMap<(String, String), Cell> = BTreeMap::new();

    for (i, path) in files.iter().take(n).enumerate() {
        let v: serde_json::Value = serde_json::from_str(&std::fs::read_to_string(path).unwrap()).unwrap();
        let genome: CreatureGenome = serde_json::from_value(v["genome"].clone()).unwrap();
        let base = battery.signature(&genome, ctx.runtime, ctx.shared_memory_decay_rate);
        let reachable = mesh_reachable_nodes(&genome);
        let executed = battery.executed_indices(&genome, ctx.runtime, ctx.shared_memory_decay_rate);
        let targets = TargetSets::new(&reachable, &executed);
        let seed_offset = 1_000_000 * (i as u64 + 1);
        for trial in 0..trials {
            let seed = seed_offset + INPUT_REF_SEED_BASE + u64::from(trial);
            let mut rng = SmallRng::seed_from_u64(seed);
            let mut selector = targets.selector(config.mutation.reachable_bias.input_ref, config.mutation.executed_bias);
            let mut mutated = genome.clone();
            let result = InputRefMutator::apply_with_food_type_count(
                &mut mutated,
                InputRefOperator::Swap,
                &mut selector,
                &mut rng,
                &config.mutation,
                food_type_count,
            );
            if result.is_err() {
                skipped += 1;
                continue;
            }
            applied += 1;
            let mut diff = None;
            for (node_before, node_after) in genome.nodes.iter().zip(&mutated.nodes) {
                for (old, new) in node_before.input_refs.iter().zip(&node_after.input_refs) {
                    if old != new {
                        assert!(diff.is_none(), "more than one entry changed");
                        diff = Some((old.clone(), new.clone()));
                    }
                }
            }
            let Some((old, new)) = diff else {
                identical += 1;
                continue;
            };
            let old_kind = input_ref_kind(&old, &config.mutation);
            let new_kind = input_ref_kind(&new, &config.mutation);
            let signature = battery.signature(&mutated, ctx.runtime, ctx.shared_memory_decay_rate);
            let class = classify(&base, &signature).class;
            let bucket = if old_kind == new_kind { &mut within } else { &mut cross };
            let key = (
                format!("{:?},{}", old_kind.class, old_kind.width),
                format!("{:?},{}", new_kind.class, new_kind.width),
            );
            let pair = by_pair.entry(key).or_default();
            match class {
                Class::Silent => { bucket.silent += 1; pair.silent += 1; }
                Class::Changed => { bucket.changed += 1; pair.changed += 1; }
                Class::Dead => { bucket.dead += 1; pair.dead += 1; }
            }
        }
    }
    let changed_total = within.changed + cross.changed;
    let c = f64::from(cross.changed) / f64::from(changed_total);
    let b = 0.125 + 0.259 * c;
    println!("{}", serde_json::json!({
        "genomes": n.min(files.len()), "trials_per_genome": trials, "food_type_count": food_type_count,
        "applied": applied, "skipped": skipped, "identical_redraw": identical,
        "within": {"silent": within.silent, "changed": within.changed, "dead": within.dead},
        "cross": {"silent": cross.silent, "changed": cross.changed, "dead": cross.dead},
        "silent_frac": f64::from(within.silent + cross.silent + identical) / f64::from(applied),
        "changed_frac": f64::from(changed_total) / f64::from(applied),
        "dead_frac": f64::from(within.dead + cross.dead) / f64::from(applied),
        "c": c, "B": b,
    }));
    for ((old, new), cell) in &by_pair {
        println!("{}", serde_json::json!({"old": old, "new": new, "silent": cell.silent, "changed": cell.changed, "dead": cell.dead}));
    }
}
```

## T13.F05 recruitment-path family after the within-kind `Swap`

`crates/v3-core/src/neighborhood/recruitment_paths/qualification.rs`: the
two unprepared forms opened with a cross-kind `InputRef.Swap` (`FoodHere` to
